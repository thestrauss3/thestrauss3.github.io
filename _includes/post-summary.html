{%- comment -%} A post's excerpt for the lists: its `summary` front matter exactly as written when set, otherwise the first one or two sentences of its first paragraph. {%- endcomment -%} {%- assign p = include.post -%} {%- if p.summary and p.summary != "" -%} {{- p.summary | escape | strip -}} {%- else -%} {%- capture nl %} {% endcapture -%} {%- assign text = p.excerpt | strip_html | replace: nl, " " | replace: " ", " " | strip -%} {%- assign sentences = text | replace: ". ", ".§" | replace: "? ", "?§" | replace: "! ", "!§" | split: "§" -%} {%- assign out = sentences[0] -%} {%- if sentences.size > 1 and out.size < 140 -%} {%- assign out = out | append: " " | append: sentences[1] -%} {%- endif -%} {%- if out.size > 320 -%}{%- assign out = out | truncate: 300 -%}{%- endif -%} {{- out -}} {%- endif -%}
